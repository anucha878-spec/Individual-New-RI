<![CDATA[/* เคสใหม่ จะดึงตาม ช่วงเวลาที่กำหนด Parameter 1. :policyNo = Input policyNo 2. :targetStartDate = Input targetStartDate แปลงเป็น number และใช้ format YYYYMMDD โดย YYYY คือปี พ.ศ. เช่น 25680705 3. :targetEndDate = Input targetEndDate แปลงเป็น number และใช้ format YYYYMMDD โดย YYYY คือปี พ.ศ. เช่น 25690101 4. :idCard = Input idCard ความหมายข้อมูลที่ดึง POLICY_NO = เลขที่กรมธรรม์ */ -- อุตสาหกรรม ปช. SELECT &#39;I&#39; AS POLICY_TYPE, A.MBPOL# AS POLICY_NO FROM ILISLIB.POLCYMB0 A INNER JOIN ILISLIB.CUSTMMC0 B ON A.MBCSID = B.MCCSID INNER JOIN ( -- ค้นหาเบี้ยและค่าคอม ของอุตสาหกรรม SELECT policy --,RECPNO,RCACDT,RCMODE,(FSTPRM-FSTDIS) AS SUM_PREMIUM ,RCCOMM FROM ILISLIB.RECEIPTS A WHERE RCSTST=&#39;U&#39; -- U : ใบเสร็จที่ ชำระแล้วของ เคสใหม่ ) RECP ON A.MBPOL# = RECP.POLICY WHERE 1 = 1 AND A.MBPOL# &lt;&gt; :policyNo AND (A.MBCDTE BETWEEN :targetStartDate AND :targetEndDate) AND B.MCIDNO = :idCard ]]>
