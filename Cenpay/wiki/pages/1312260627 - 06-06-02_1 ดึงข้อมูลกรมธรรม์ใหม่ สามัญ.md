<![CDATA[/*เคสใหม่ จะดึงตาม ช่วงเวลาที่กำหนด Parameter 1. :policyNo = Input policyNo 2. :targetStartDate = Input targetStartDate แปลงเป็น number และใช้ format YYYYMMDD โดย YYYY คือปี พ.ศ. เช่น 25680705 3. :targetEndDate = Input targetEndDate แปลงเป็น number และใช้ format YYYYMMDD โดย YYYY คือปี พ.ศ. เช่น 25690101 4. :idCard = Input idCard ความหมายข้อมูลที่ดึง POLICY_NO = เลขที่กรมธรรม์ */ -- สามัญ * ต้องแปลงวันที่จาก 6 เป้น 8 หลัก ก่อนใช้งาน SELECT &#39;O&#39; AS POLICY_TYPE, A.POLIC# AS POLICY_NO FROM OLIS.OLPPOLMS A INNER JOIN OLIS.OLPCUSMS B ON A.CUSTM# = B.CUSTM# INNER JOIN ( -- ค้นหาข้อมูลเบี้ยจากข้อมูลใบเสร็จ SELECT RCPOL#,RC# --,(RCPYDT + 25000000) AS RCPYDT --,(RCPOPR+RD13PR+RD24PR+RD24PR+RDHCPR+RDOTPR+RCEXPR) AS SUM_PREMIUM FROM OLIS.OLPRECDT WHERE RCPYY# =&#39;1&#39; -- ชำระปีที่ 1 AND RCPYI# =&#39;1&#39; -- ชำระ งวดที่ 1 ) RECP ON A.POLIC# = RECP.RCPOL# INNER JOIN ( -- ค้นหาข้อมูลค่าคอมมิชชั่น SELECT RCPOL#,RC# -- ,SUM(CMAMTR) AS COMMISSION FROM OLIS.OLPRECIC GROUP BY RCPOL#,RC# ) COMM ON COMM.RCPOL# = A.POLIC# AND RECP.RC# = COMM.RC# WHERE 1 = 1 AND A.POLIC# &lt;&gt; :policyNo AND ((A.POCMDT + 25000000) BETWEEN :targetStartDate AND :targetEndDate) AND B.CUIDNO = :idCard]]>
