## Process

**ค้นหาข้อมูลรายการเปลี่ยนแปลงข้อมูลที่รออนุมัติที่ระบบ CIS DB: CIS**
<![CDATA[select h.id as referenceNo ,array_to_string(array_agg(distinct dg.group_desc), &#39;, &#39;) as changeTitle ,h.create_date as createDate ,h.create_by as createBy ,hp.process_branch_name as branchName ,h.last_status as lastStatus ,st.status_desc as statusDesc from cis_update_history h -- ข้อมูลใบข้อแก้ไข inner join cis_update_history_detail hd on ( hd.reference_no = h.id) -- ข้อมูลรายละเอียดการแก้ไข inner join cis_update_history_process hp on ( hp.reference_no = h.id and hp.process_status = &#39;Request&#39;) inner join cis_data_group dg on(dg.group_id = hd.data_group_code) -- Master กลุ่มข้อมูล inner join cis_lookup_edit_status st on h.last_status = st.status_name -- Master สถานะ where h.last_status in(&#39;Request&#39;,&#39;Reverted&#39;) and hd.data_group_code in :dataGroupCode and h.customer_id = :customerID and case when hd.policy_no is not null then hd.policy_no = :policyNo else 1=1 end group by h.id, hp.process_branch_name, st.status_desc;]]>
